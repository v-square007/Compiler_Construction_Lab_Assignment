Node* createNode(const char* label, const char* value) {
    Node* n = (Node*)malloc(sizeof(Node));
    if (!n) return NULL;

    strncpy(n->label, label ? label : "", 49);
    n->label[49] = '\0';

    if (value) {
        strncpy(n->value, value, 49);
        n->value[49] = '\0';
    } else {
        n->value[0] = '\0';
    }

    n->line = line_number;
    n->child_count = 0;

    for (int i = 0; i < 10; i++) {
        n->children[i] = NULL;
    }

    return n;
}

void addChild(Node* parent, Node* child) {
    if (parent && child && parent->child_count < 10) {
        parent->children[parent->child_count++] = child;
    }
}

void print_tree_visual(Node* node, char* prefix, int is_last) {
    if (!node) return;

    printf("%s", prefix);
    printf(is_last ? "+-- " : "|-- ");

    if (node->value[0] != '\0')
        printf("%s: [%s] (line %d)\n", node->label, node->value, node->line);
    else
        printf("%s (line %d)\n", node->label, node->line);

    char new_prefix[MAX_TOKENS];
    sprintf(new_prefix, "%s%s", prefix, is_last ? "    " : "|   ");

    for (int i = 0; i < node->child_count; i++) {
        print_tree_visual(node->children[i], new_prefix,
                          i == node->child_count - 1);
    }
}

void leftmost_derivation(Node* root) {
    if (!root) return;

    Node* sentential[MAX_TOKENS];
    int count = 1, step = 1;
    sentential[0] = root;

    while (1) {
        printf("Step %d: ", step++);
        for (int i = 0; i < count; i++) {
            printf("%s ",
                   sentential[i]->value[0] ?
                   sentential[i]->value : sentential[i]->label);
        }
        printf("\n");

        int pos = -1;
        for (int i = 0; i < count; i++) {
            if (sentential[i]->child_count > 0) {
                pos = i;
                break;
            }
        }
        if (pos == -1) break;

        Node* temp[MAX_TOKENS];
        int idx = 0;

        for (int i = 0; i < pos; i++) temp[idx++] = sentential[i];
        for (int i = 0; i < sentential[pos]->child_count; i++)
            temp[idx++] = sentential[pos]->children[i];
        for (int i = pos + 1; i < count; i++) temp[idx++] = sentential[i];

        count = idx;
        for (int i = 0; i < count; i++) sentential[i] = temp[i];
    }
}

void rightmost_derivation(Node* root) {
    if (!root) return;

    Node* sentential[MAX_TOKENS];
    int count = 1, step = 1;
    sentential[0] = root;

    while (1) {
        printf("Step %d: ", step++);
        for (int i = 0; i < count; i++) {
            printf("%s ",
                   sentential[i]->value[0] ?
                   sentential[i]->value : sentential[i]->label);
        }
        printf("\n");

        int pos = -1;
        for (int i = count - 1; i >= 0; i--) {
            if (sentential[i]->child_count > 0) {
                pos = i;
                break;
            }
        }
        if (pos == -1) break;

        Node* temp[MAX_TOKENS];
        int idx = 0;

        for (int i = 0; i < pos; i++) temp[idx++] = sentential[i];
        for (int i = 0; i < sentential[pos]->child_count; i++)
            temp[idx++] = sentential[pos]->children[i];
        for (int i = pos + 1; i < count; i++) temp[idx++] = sentential[i];

        count = idx;
        for (int i = 0; i < count; i++) sentential[i] = temp[i];
    }
}

Node* find_assignment(Node* node) {
    if (!node) return NULL;
    if (strcmp(node->label, "assign_stmt") == 0) return node;

    for (int i = 0; i < node->child_count; i++) {
        Node* r = find_assignment(node->children[i]);
        if (r) return r;
    }
    return NULL;
}

/* ================== ERROR HANDLING ==================== */

static void record_error(const char* msg, const char* tok, int line) {
    if (error_count >= MAX_ERRORS) return;

    if (error_count > 0) {
        SyntaxError* prev = &error_list[error_count - 1];
        if (prev->line == line &&
            strcmp(prev->token, tok ? tok : "") == 0) {
            return;
        }
    }

    SyntaxError* e = &error_list[error_count++];
    e->line = line;

    strncpy(e->token, tok ? tok : "", 63);
    e->token[63] = '\0';

    strncpy(e->message, msg ? msg : "", 255);
    e->message[255] = '\0';

    if (strstr(msg, "expecting") && strstr(msg, "';'"))
        strncpy(e->type, "Missing Semicolon", 63);
    else if (strstr(msg, "expecting") && strstr(msg, "')'"))
        strncpy(e->type, "Missing Closing Parenthesis", 63);
    else if (strstr(msg, "expecting") && strstr(msg, "'('"))
        strncpy(e->type, "Missing Opening Parenthesis", 63);
    else if (strstr(msg, "expecting") && strstr(msg, "'}'"))
        strncpy(e->type, "Missing Closing Brace", 63);
    else if (strstr(msg, "expecting") && strstr(msg, "'{'"))
        strncpy(e->type, "Missing Opening Brace", 63);
    else if (strstr(msg, "expecting") && strstr(msg, "'='"))
        strncpy(e->type, "Missing Assignment Operator", 63);
    else if (strstr(msg, "unexpected '}'"))
        strncpy(e->type, "Unexpected Closing Brace", 63);
    else if (strstr(msg, "unexpected '{'"))
        strncpy(e->type, "Unexpected Opening Brace", 63);
    else if (strstr(msg, "unexpected ')'"))
        strncpy(e->type, "Unexpected Closing Parenthesis", 63);
    else if (strstr(msg, "unexpected ELSE"))
        strncpy(e->type, "Unexpected else (no matching if)", 63);
    else if (strstr(msg, "end of file"))
        strncpy(e->type, "Unexpected End of File", 63);
    else if (strstr(msg, "unexpected"))
        strncpy(e->type, "Unexpected Token", 63);
    else
        strncpy(e->type, "Syntax Error", 63);

    e->type[63] = '\0';
}

void yyerror(const char* s) {
    record_error(s, yytext, line_number);
}

static void print_error_summary(void) {
    printf("\n========================================\n");
    printf("  ERROR SUMMARY  (%d error%s found)\n",
           error_count, error_count == 1 ? "" : "s");
    printf("========================================\n");
    printf("%-4s  %-40s  %-6s  %s\n",
           "No.", "Error Type", "Line", "Details");
    printf("%-4s  %-40s  %-6s  %s\n",
           "---", "----------", "----", "-------");

    for (int i = 0; i < error_count; i++) {
        printf("%-4d  %-40s  %-6d  %s (near '%s')\n",
               i + 1,
               error_list[i].type,
               error_list[i].line,
               error_list[i].message,
               error_list[i].token);
    }
    printf("========================================\n");
}

/* ================= SYMBOL TABLE ================= */

void reset_semantic_state(void) {
    symbol_count = 0;
    sem_error_count = 0;
    current_scope = 0;
    next_scope_id = 0;
    scope_top = -1;

    for (int i = 0; i < MAX_SCOPE_DEPTH; i++)
        scope_offsets[i] = 0;
}

void add_semantic_error(int line, const char* msg) {
    if (sem_error_count >= MAX_SEM_ERRORS) return;

    if (sem_error_count > 0) {
        SemanticError* prev = &sem_errors[sem_error_count - 1];
        if (prev->line == line && strcmp(prev->message, msg) == 0) {
            return;
        }
    }

    sem_errors[sem_error_count].line = line;
    strncpy(sem_errors[sem_error_count].message, msg, 255);
    sem_errors[sem_error_count].message[255] = '\0';
    sem_error_count++;
}

void enter_scope(void) {
    if (scope_top < MAX_SCOPE_DEPTH - 1) {
        scope_stack[++scope_top] = current_scope;
        current_scope = ++next_scope_id;
        scope_offsets[current_scope] = 0;
        printf("\n>> Enter Scope %d\n", current_scope);
    }
}

void exit_scope(void) {
    printf("<< Exit Scope %d\n", current_scope);
    if (scope_top >= 0) {
        current_scope = scope_stack[scope_top--];
    }
}

int get_type_size(const char* type) {
    if (strcmp(type, "int") == 0) return 4;
    if (strcmp(type, "float") == 0) return 8;
    if (strcmp(type, "bool") == 0) return 1;
    return 4;
}

Symbol* lookup_current_scope(const char* name) {
    printf("\n[LOOKUP CURRENT SCOPE %d] '%s'\n", current_scope, name);

    for (int i = symbol_count - 1; i >= 0; i--) {
        if (strcmp(symbol_table[i].name, name) == 0 &&
            symbol_table[i].scope == current_scope) {

            printf("  ✔ Found in current scope\n");
            return &symbol_table[i];
        }
    }

    printf("  ✘ Not found in current scope\n");
    return NULL;
}

Symbol* lookup_symbol(const char* name) {
    printf("\n[LOOKUP] Searching for '%s'\n", name);

    int active_scopes[MAX_SCOPE_DEPTH];
    int count = 0;

    /* Collect scopes: current → outer */
    active_scopes[count++] = current_scope;
    for (int i = scope_top; i >= 0; i--) {
        active_scopes[count++] = scope_stack[i];
    }

    for (int s = 0; s < count; s++) {
        int sc = active_scopes[s];
        printf("  Checking Scope %d...\n", sc);

        for (int i = symbol_count - 1; i >= 0; i--) {
            if (strcmp(symbol_table[i].name, name) == 0 &&
                symbol_table[i].scope == sc) {

                printf("  ✔ Found '%s' in Scope %d (Type=%s, Offset=%d)\n",
                       name,
                       sc,
                       symbol_table[i].type,
                       symbol_table[i].offset);

                return &symbol_table[i];
            }
        }
    }

    printf("  ✘ '%s' NOT FOUND in any active scope\n", name);
    return NULL;
}

void insert_symbol(const char* name, const char* type, int line) {
    if (symbol_count >= MAX_SYMBOLS) return;

    if (lookup_current_scope(name) != NULL) {
        char msg[256];
        snprintf(msg, sizeof(msg),
                 "Multiple declaration of variable '%s' in same scope", name);
        add_semantic_error(line, msg);
        return;
    }

    Symbol* s = &symbol_table[symbol_count++];
    strncpy(s->name, name, 63);
    s->name[63] = '\0';

    strncpy(s->type, type, 31);
    s->type[31] = '\0';

    s->scope = current_scope;
    s->offset = scope_offsets[current_scope];
    s->line = line;

    scope_offsets[current_scope] += get_type_size(type);

    printf("Inserted -> Name: %-10s Type: %-8s Scope: %-3d Offset: %-3d\n",
           s->name, s->type, s->scope, s->offset);
}

void print_symbol_table(void) {
    printf("\n================ SYMBOL TABLE ================\n");
    printf("%-5s %-15s %-10s %-10s %-10s %-10s\n",
           "No", "Name", "Type", "Scope", "Offset", "Line");
    printf("---------------------------------------------------------------\n");

    for (int i = 0; i < symbol_count; i++) {
        printf("%-5d %-15s %-10s %-10d %-10d %-10d\n",
               i + 1,
               symbol_table[i].name,
               symbol_table[i].type,
               symbol_table[i].scope,
               symbol_table[i].offset,
               symbol_table[i].line);
    }

    printf("==============================================================\n");
}

void print_semantic_errors(void) {
    printf("\n================ SEMANTIC ERRORS ================\n");

        printf("No semantic errors found.\n");
    } else {

            printf("%d. Line %d: %s\n",

        }
    }
    printf("=================================================\n");
}

/* ================= TYPE INFERENCE ================= */

Node* get_first_child_with_label(Node* node, const char* label) {
    if (!node) return NULL;
    for (int i = 0; i < node->child_count; i++) {
        if (strcmp(node->children[i]->label, label) == 0)
            return node->children[i];
    }
    return NULL;
}

Node* find_condition_node(Node* node) {
    if (!node) return NULL;

    for (int i = 0; i < node->child_count; i++) {
        if (strcmp(node->children[i]->label, "bool_expr") == 0 ||
            strcmp(node->children[i]->label, "expr") == 0 ||
            strcmp(node->children[i]->label, "error_cond") == 0) {
            return node->children[i];
        }
    }

    return NULL;
}

const char* infer_expr_type(Node* node);
const char* infer_term_type(Node* node);
const char* infer_factor_type(Node* node);
const char* infer_bool_expr_type(Node* node);
const char* infer_bool_term_type(Node* node);
const char* infer_bool_atom_type(Node* node);

const char* infer_factor_type(Node* node) {
    if (!node) return "unknown";

    if (strcmp(node->label, "ID") == 0) {
        Symbol* s = lookup_symbol(node->value);
        if (!s) {
            char msg[256];
            snprintf(msg, sizeof(msg),
                     "Use of undeclared variable '%s'", node->value);
            add_semantic_error(node->line, msg);
            return "unknown";
        }
        return s->type;
    }

    if (strcmp(node->label, "INT_CONST") == 0 || strcmp(node->label, "Int") == 0)
        return "int";
    if (strcmp(node->label, "FLOAT_CONST") == 0 || strcmp(node->label, "Float") == 0)
        return "float";
    if (strcmp(node->label, "BOOL_CONST") == 0)
        return "bool";

    for (int i = 0; i < node->child_count; i++) {
        const char* t = infer_factor_type(node->children[i]);
        if (strcmp(t, "unknown") != 0) return t;
    }

    return "unknown";
}

const char* infer_term_type(Node* node) {
    if (!node) return "unknown";

    if (node->child_count == 1)
        return infer_factor_type(node->children[0]);

    if (node->child_count == 3) {
        const char* left = infer_term_type(node->children[0]);
        const char* right = infer_factor_type(node->children[2]);

        if ((!strcmp(left, "int") || !strcmp(left, "float")) &&
            (!strcmp(right, "int") || !strcmp(right, "float"))) {
            if (!strcmp(left, "float") || !strcmp(right, "float"))
                return "float";
            return "int";
        }

        add_semantic_error(node->line, "Type mismatch in term expression");
        return "unknown";
    }

    return "unknown";
}

const char* infer_expr_type(Node* node) {
    if (!node) return "unknown";

    if (node->child_count == 1)
        return infer_term_type(node->children[0]);

    if (node->child_count == 3) {
        const char* left = infer_expr_type(node->children[0]);
        const char* right = infer_term_type(node->children[2]);

        if ((!strcmp(left, "int") || !strcmp(left, "float")) &&
            (!strcmp(right, "int") || !strcmp(right, "float"))) {
            if (!strcmp(left, "float") || !strcmp(right, "float"))
                return "float";
            return "int";
        }

        add_semantic_error(node->line, "Type mismatch in arithmetic expression");
        return "unknown";
    }

    return infer_factor_type(node);
}

const char* infer_bool_atom_type(Node* node) {
    if (!node) return "unknown";

    if (node->child_count == 3) {
        if (strcmp(node->children[1]->label, "REL") == 0) {
            const char* left = infer_expr_type(node->children[0]);
            const char* right = infer_expr_type(node->children[2]);

            if ((!strcmp(left, "int") || !strcmp(left, "float")) &&
                (!strcmp(right, "int") || !strcmp(right, "float"))) {
                return "bool";
            }

            add_semantic_error(node->line,
                               "Type mismatch in relational expression");
            return "unknown";
        }

        return infer_bool_expr_type(node->children[1]);
    }

    if (node->child_count == 1) {
        if (strcmp(node->children[0]->label, "BOOL_CONST") == 0)
            return "bool";
    }

    for (int i = 0; i < node->child_count; i++) {
        if (strcmp(node->children[i]->label, "bool_expr") == 0)
            return infer_bool_expr_type(node->children[i]);
    }

    return "unknown";
}

const char* infer_bool_term_type(Node* node) {
    if (!node) return "unknown";

    if (node->child_count == 1) {
        if (strcmp(node->children[0]->label, "bool_atom") == 0)
            return infer_bool_atom_type(node->children[0]);
    }

    if (node->child_count == 2) {
        const char* t = infer_bool_term_type(node->children[1]);
        if (strcmp(t, "bool") != 0) {
            add_semantic_error(node->line, "Invalid boolean condition");
            return "unknown";
        }
        return "bool";
    }

    return "unknown";
}

const char* infer_bool_expr_type(Node* node) {
    if (!node) return "unknown";

    if (node->child_count == 1)
        return infer_bool_term_type(node->children[0]);

    if (node->child_count == 3) {
        const char* left = infer_bool_expr_type(node->children[0]);
        const char* right = infer_bool_term_type(node->children[2]);

        if (strcmp(left, "bool") == 0 && strcmp(right, "bool") == 0)
            return "bool";

        add_semantic_error(node->line, "Invalid boolean condition");
        return "unknown";
    }

    return "unknown";
}

/* ================= SEMANTIC TRAVERSAL ================= */

void semantic_traverse(Node* node) {
    if (!node) return;

    if (strcmp(node->label, "block") == 0) {
        enter_scope();
        for (int i = 0; i < node->child_count; i++)
            semantic_traverse(node->children[i]);
        exit_scope();
        return;
    }

    if (strcmp(node->label, "decl_stmt") == 0) {
        Node* typeNode = get_first_child_with_label(node, "TYPE");
        Node* idNode = get_first_child_with_label(node, "ID");

        if (typeNode && idNode) {
            insert_symbol(idNode->value, typeNode->value, idNode->line);
        }

        Node* exprNode = NULL;
        for (int i = 0; i < node->child_count; i++) {
            if (strcmp(node->children[i]->label, "expr") == 0) {
                exprNode = node->children[i];
                break;
            }
        }

        if (typeNode && exprNode) {
            const char* rhs_type = infer_expr_type(exprNode);

            if (strcmp(rhs_type, "unknown") != 0 &&
                strcmp(typeNode->value, rhs_type) != 0) {
                if (!(strcmp(typeNode->value, "float") == 0 &&
                      strcmp(rhs_type, "int") == 0)) {
                    char msg[256];
                    snprintf(msg, sizeof(msg),
                             "Type mismatch in declaration of '%s' (lhs=%s, rhs=%s)",
                             idNode ? idNode->value : "?",
                             typeNode->value, rhs_type);
                    add_semantic_error(node->line, msg);
                }
            }
        }
    }
    else if (strcmp(node->label, "assign_stmt") == 0) {
        Node* idNode = get_first_child_with_label(node, "ID");
        Node* exprNode = NULL;

        for (int i = 0; i < node->child_count; i++) {
            if (strcmp(node->children[i]->label, "expr") == 0) {
                exprNode = node->children[i];
                break;
            }
        }

        if (idNode) {
            Symbol* s = lookup_symbol(idNode->value);
            if (!s) {
                char msg[256];
                snprintf(msg, sizeof(msg),
                         "Use of undeclared variable '%s'", idNode->value);
                add_semantic_error(idNode->line, msg);
            } else if (exprNode) {
                const char* rhs_type = infer_expr_type(exprNode);

                if (strcmp(rhs_type, "unknown") != 0 &&
                    strcmp(s->type, rhs_type) != 0) {
                    if (!(strcmp(s->type, "float") == 0 &&
                          strcmp(rhs_type, "int") == 0)) {
                        char msg[256];
                        snprintf(msg, sizeof(msg),
                                 "Type mismatch in assignment to '%s' (lhs=%s, rhs=%s)",
                                 idNode->value, s->type, rhs_type);
                        add_semantic_error(node->line, msg);
                    }
                }
            }
        }
    }
    else if (strcmp(node->label, "if_stmt") == 0) {
    Node* condNode = find_condition_node(node);

    if (condNode) {
        if (strcmp(condNode->label, "error_cond") == 0) {
            add_semantic_error(condNode->line,
                "Invalid boolean condition in if statement");
        } else {
            const char* cond_type;

            if (strcmp(condNode->label, "bool_expr") == 0) {
                cond_type = infer_bool_expr_type(condNode);
            } else {
                cond_type = "not_bool";
            }

            if (strcmp(cond_type, "bool") != 0) {
                add_semantic_error(condNode->line,
                    "Invalid boolean condition in if statement");
            }
        }
    }
}
    else if (strcmp(node->label, "while_stmt") == 0) {
    Node* condNode = find_condition_node(node);

    if (condNode) {
        if (strcmp(condNode->label, "error_cond") == 0) {
            add_semantic_error(condNode->line,
                "Invalid boolean condition in while statement");
        } else {
            const char* cond_type;

            if (strcmp(condNode->label, "bool_expr") == 0) {
                cond_type = infer_bool_expr_type(condNode);
            } else {
                cond_type = "not_bool";
            }

            if (strcmp(cond_type, "bool") != 0) {
                add_semantic_error(condNode->line,
                    "Invalid boolean condition in while statement");
            }
        }
    }
}

    for (int i = 0; i < node->child_count; i++)
        semantic_traverse(node->children[i]);
}



/* ================= TAC GENERATION ================= */

void reset_tac_state(void) {
    tac_count = 0;
    temp_count = 0;
    label_count = 0;
}

char* new_temp(void) {
    if (temp_count >= MAX_TAC) return NULL;
    snprintf(temp_names[temp_count], sizeof(temp_names[temp_count]), "t%d", temp_count + 1);
    return temp_names[temp_count++];
}

char* new_label(void) {
    if (label_count >= MAX_TAC) return NULL;
    snprintf(label_names[label_count], sizeof(label_names[label_count]), "L%d", label_count + 1);
    return label_names[label_count++];
}

void emit_quad(const char* op, const char* arg1, const char* arg2, const char* result) {
    if (tac_count >= MAX_TAC) return;
    tac_table[tac_count].index = tac_count;
    snprintf(tac_table[tac_count].op, sizeof(tac_table[tac_count].op), "%s", op ? op : "");
    snprintf(tac_table[tac_count].arg1, sizeof(tac_table[tac_count].arg1), "%s", arg1 ? arg1 : "-");
    snprintf(tac_table[tac_count].arg2, sizeof(tac_table[tac_count].arg2), "%s", arg2 ? arg2 : "-");
    snprintf(tac_table[tac_count].result, sizeof(tac_table[tac_count].result), "%s", result ? result : "-");
    tac_count++;
}

static Node* unwrap_single(Node* node) {
    while (node && (strcmp(node->label, "expr") == 0 || strcmp(node->label, "term") == 0 ||
                    strcmp(node->label, "factor") == 0 || strcmp(node->label, "bool_expr") == 0 ||
                    strcmp(node->label, "bool_term") == 0 || strcmp(node->label, "bool_atom") == 0) &&
           node->child_count == 1) {
        node = node->children[0];
    }
    return node;
}

static int is_wrapper_label(const char* s) {
    return strcmp(s, "expr") == 0 || strcmp(s, "term") == 0 || strcmp(s, "factor") == 0 ||
           strcmp(s, "bool_expr") == 0 || strcmp(s, "bool_term") == 0 || strcmp(s, "bool_atom") == 0;
}

char* generate_expr_tac(Node* node) {
    if (!node) return strdup("?");
    node = unwrap_single(node);

    if (!node) return strdup("?");

    if (strcmp(node->label, "ID") == 0 || strcmp(node->label, "INT_CONST") == 0 ||
        strcmp(node->label, "FLOAT_CONST") == 0 || strcmp(node->label, "BOOL_CONST") == 0) {
        return strdup(node->value);
    }

    if (strcmp(node->label, "factor") == 0 && node->child_count == 3 &&
        strcmp(node->children[0]->label, "Delimiter") == 0) {
        return generate_expr_tac(node->children[1]);
    }

    if ((strcmp(node->label, "expr") == 0 || strcmp(node->label, "term") == 0) && node->child_count == 3) {
        char* left = generate_expr_tac(node->children[0]);
        char* right = generate_expr_tac(node->children[2]);
        char* temp = new_temp();
        emit_quad(node->children[1]->value, left, right, temp);
        return strdup(temp);
    }

    if (node->child_count == 1 && is_wrapper_label(node->label))
        return generate_expr_tac(node->children[0]);

    for (int i = 0; i < node->child_count; i++) {
        if (is_wrapper_label(node->children[i]->label) || strcmp(node->children[i]->label, "ID") == 0 ||
            strcmp(node->children[i]->label, "INT_CONST") == 0 || strcmp(node->children[i]->label, "FLOAT_CONST") == 0)
            return generate_expr_tac(node->children[i]);
    }

    return strdup("?");
}

char* generate_bool_tac(Node* node) {
    if (!node) return strdup("?");
    node = unwrap_single(node);

    if (!node) return strdup("?");

    if (strcmp(node->label, "bool_term") == 0 && node->child_count == 2 &&
        strcmp(node->children[0]->label, "NOT") == 0) {
        char* inner = generate_bool_tac(node->children[1]);
        char* temp = new_temp();
        emit_quad("!", inner, "-", temp);
        return strdup(temp);
    }

    if (strcmp(node->label, "bool_atom") == 0 && node->child_count == 3 &&
        strcmp(node->children[1]->label, "REL") == 0) {
        char* left = generate_expr_tac(node->children[0]);
        char* right = generate_expr_tac(node->children[2]);
        char* temp = new_temp();
        emit_quad(node->children[1]->value, left, right, temp);
        return strdup(temp);
    }

    if (strcmp(node->label, "bool_expr") == 0 && node->child_count == 3 &&
        strcmp(node->children[1]->label, "LOGIC") == 0) {
        char* left = generate_bool_tac(node->children[0]);
        char* right = generate_bool_tac(node->children[2]);
        char* temp = new_temp();
        emit_quad(node->children[1]->value, left, right, temp);
        return strdup(temp);
    }

    if (strcmp(node->label, "bool_atom") == 0 && node->child_count == 1)
        return generate_bool_tac(node->children[0]);

    if (is_wrapper_label(node->label) && node->child_count == 1)
        return generate_bool_tac(node->children[0]);

    return generate_expr_tac(node);
}

static Node* get_stmt_child(Node* node, int idx) {
    int seen = -1;
    for (int i = 0; i < node->child_count; i++) {
        Node* c = node->children[i];
        if (!c) continue;
        if (strcmp(c->label, "IF") == 0 || strcmp(c->label, "ELSE") == 0 || strcmp(c->label, "WHILE") == 0 ||
            strcmp(c->label, "Assign") == 0 || strcmp(c->label, "Delimiter") == 0 || strcmp(c->label, "PRINT") == 0)
            continue;
        seen++;
        if (seen == idx) return c;
    }
    return NULL;
}

void generate_stmt_list_tac(Node* node) {
    if (!node) return;
    if (strcmp(node->label, "stmt_list") != 0) {
        generate_stmt_tac(node);
        return;
    }
    for (int i = 0; i < node->child_count; i++)
        generate_stmt_list_tac(node->children[i]);
}

void generate_stmt_tac(Node* node) {
    if (!node) return;

    if (strcmp(node->label, "program") == 0) {
        for (int i = 0; i < node->child_count; i++) generate_stmt_tac(node->children[i]);
        return;
    }

    if (strcmp(node->label, "stmt_list") == 0) {
        generate_stmt_list_tac(node);
        return;
    }

    if (strcmp(node->label, "block") == 0) {
        for (int i = 0; i < node->child_count; i++) generate_stmt_tac(node->children[i]);
        return;
    }

    if (strcmp(node->label, "decl_stmt") == 0) {
        Node* idNode = get_first_child_with_label(node, "ID");
        Node* exprNode = get_first_child_with_label(node, "expr");
        if (idNode && exprNode) {
            char* rhs = generate_expr_tac(exprNode);
            emit_quad("=", rhs, "-", idNode->value);
        }
        return;
    }

    if (strcmp(node->label, "assign_stmt") == 0) {
        Node* idNode = get_first_child_with_label(node, "ID");
        Node* exprNode = get_first_child_with_label(node, "expr");
        if (idNode && exprNode) {
            char* rhs = generate_expr_tac(exprNode);
            emit_quad("=", rhs, "-", idNode->value);
        }
        return;
    }

    if (strcmp(node->label, "print_stmt") == 0) {
        Node* exprNode = get_first_child_with_label(node, "expr");
        if (exprNode) {
            char* place = generate_expr_tac(exprNode);
            emit_quad("print", place, "-", "-");
        }
        return;
    }

    if (strcmp(node->label, "if_stmt") == 0) {
        Node* condNode = get_stmt_child(node, 0);
        Node* thenBlock = get_stmt_child(node, 1);
        Node* elseBlock = get_stmt_child(node, 2);

        char* cond = generate_bool_tac(condNode);
        char* ltrue = new_label();
        char* lfalse = new_label();
        char* lend = new_label();

        emit_quad("ifTrue", cond, "-", ltrue);
        emit_quad("goto", "-", "-", lfalse);
        emit_quad("label", "-", "-", ltrue);
        generate_stmt_tac(thenBlock);

        if (elseBlock) {
            emit_quad("goto", "-", "-", lend);
            emit_quad("label", "-", "-", lfalse);
            generate_stmt_tac(elseBlock);
            emit_quad("label", "-", "-", lend);
        } else {
            emit_quad("label", "-", "-", lfalse);
        }
        return;
    }

    if (strcmp(node->label, "while_stmt") == 0) {
        Node* condNode = get_stmt_child(node, 0);
        Node* bodyBlock = get_stmt_child(node, 1);

        char* lstart = new_label();
        char* lbody = new_label();
        char* lend = new_label();

        emit_quad("label", "-", "-", lstart);
        char* cond = generate_bool_tac(condNode);
        emit_quad("ifTrue", cond, "-", lbody);
        emit_quad("goto", "-", "-", lend);
        emit_quad("label", "-", "-", lbody);
        generate_stmt_tac(bodyBlock);
        emit_quad("goto", "-", "-", lstart);
        emit_quad("label", "-", "-", lend);
        return;
    }
}

void generate_program_tac(Node* root) {
    reset_tac_state();
    generate_stmt_tac(root);
}

/* ================= TAC OPTIMIZATION + TARGET CODE ================= */

static int is_integer_const(const char* s) {
    int i = 0;
    if (!s || !s[0] || strcmp(s, "-") == 0) return 0;
    if (s[0] == '-') i = 1;
    for (; s[i]; i++) {
        if (s[i] < '0' || s[i] > '9') return 0;
    }
    return 1;
}

static int is_arithmetic_op(const char* op) {
    return strcmp(op, "+") == 0 || strcmp(op, "-") == 0 ||
           strcmp(op, "*") == 0 || strcmp(op, "/") == 0 ||
           strcmp(op, "%") == 0;
}

static int is_relational_op(const char* op) {
    return strcmp(op, "<") == 0 || strcmp(op, ">") == 0 ||
           strcmp(op, "<=") == 0 || strcmp(op, ">=") == 0 ||
           strcmp(op, "==") == 0 || strcmp(op, "!=") == 0;
}

static int is_logic_op(const char* op) {
    return strcmp(op, "&&") == 0 || strcmp(op, "||") == 0;
}

static int eval_binary_int(const char* op, int a, int b, int* ok) {
    *ok = 1;
    if (strcmp(op, "+") == 0) return a + b;
    if (strcmp(op, "-") == 0) return a - b;
    if (strcmp(op, "*") == 0) return a * b;
    if (strcmp(op, "/") == 0) {
        if (b == 0) { *ok = 0; return 0; }
        return a / b;
    }
    if (strcmp(op, "%") == 0) {
        if (b == 0) { *ok = 0; return 0; }
        return a % b;
    }
    if (strcmp(op, "<") == 0) return a < b;
    if (strcmp(op, ">") == 0) return a > b;
    if (strcmp(op, "<=") == 0) return a <= b;
    if (strcmp(op, ">=") == 0) return a >= b;
    if (strcmp(op, "==") == 0) return a == b;
    if (strcmp(op, "!=") == 0) return a != b;
    if (strcmp(op, "&&") == 0) return (a != 0) && (b != 0);
    if (strcmp(op, "||") == 0) return (a != 0) || (b != 0);
    *ok = 0;
    return 0;
}

static void replace_uses_after(int start, const char* old_name, const char* new_value) {
    for (int i = start; i < tac_count; i++) {
        if (strcmp(tac_table[i].arg1, old_name) == 0)
            snprintf(tac_table[i].arg1, sizeof(tac_table[i].arg1), "%s", new_value);
        if (strcmp(tac_table[i].arg2, old_name) == 0)
            snprintf(tac_table[i].arg2, sizeof(tac_table[i].arg2), "%s", new_value);
    }
}

void optimize_tac_constant_folding(void) {
    for (int i = 0; i < tac_count; i++) {
        Quadruple* q = &tac_table[i];

        if ((is_arithmetic_op(q->op) || is_relational_op(q->op) || is_logic_op(q->op)) &&
            is_integer_const(q->arg1) && is_integer_const(q->arg2)) {

            int a = atoi(q->arg1);
            int b = atoi(q->arg2);
            int ok = 0;
            int value = eval_binary_int(q->op, a, b, &ok);

            if (ok) {
                char folded[64];
                snprintf(folded, sizeof(folded), "%d", value);

                /* Convert: t1 = 2 * 3  into: t1 = 6 */
                snprintf(q->op, sizeof(q->op), "=");
                snprintf(q->arg1, sizeof(q->arg1), "%s", folded);
                snprintf(q->arg2, sizeof(q->arg2), "-");

                /* Replace later uses of t1 with 6. */
                replace_uses_after(i + 1, q->result, folded);
            }
        }

        if (strcmp(q->op, "!") == 0 && is_integer_const(q->arg1)) {
            int value = !atoi(q->arg1);
            char folded[64];
            snprintf(folded, sizeof(folded), "%d", value);
            snprintf(q->op, sizeof(q->op), "=");
            snprintf(q->arg1, sizeof(q->arg1), "%s", folded);
            snprintf(q->arg2, sizeof(q->arg2), "-");
            replace_uses_after(i + 1, q->result, folded);
        }
    }
}

void print_optimized_line_changes(Quadruple before[], int before_count) {
    int changed = 0;

    printf("\n================ OPTIMIZED / CHANGED TAC LINES ================\n");
    printf("%-5s %-38s %-38s\n", "No", "Before", "After");
    printf("-------------------------------------------------------------------------------------\n");

    for (int i = 0; i < tac_count && i < before_count; i++) {
        Quadruple b = before[i];
        Quadruple a = tac_table[i];

        if (strcmp(b.op, a.op) != 0 ||
            strcmp(b.arg1, a.arg1) != 0 ||
            strcmp(b.arg2, a.arg2) != 0 ||
            strcmp(b.result, a.result) != 0) {

            char before_line[512];
            char after_line[512];

            if (strcmp(b.op, "=") == 0)
                snprintf(before_line, sizeof(before_line), "%s = %s", b.result, b.arg1);
            else if (strcmp(b.op, "goto") == 0)
                snprintf(before_line, sizeof(before_line), "goto %s", b.result);
            else if (strcmp(b.op, "ifTrue") == 0)
                snprintf(before_line, sizeof(before_line), "if %s goto %s", b.arg1, b.result);
            else if (strcmp(b.op, "label") == 0)
                snprintf(before_line, sizeof(before_line), "%s:", b.result);
            else
                snprintf(before_line, sizeof(before_line), "%s = %s %s %s", b.result, b.arg1, b.op, b.arg2);

            if (strcmp(a.op, "=") == 0)
                snprintf(after_line, sizeof(after_line), "%s = %s", a.result, a.arg1);
            else if (strcmp(a.op, "goto") == 0)
                snprintf(after_line, sizeof(after_line), "goto %s", a.result);
            else if (strcmp(a.op, "ifTrue") == 0)
                snprintf(after_line, sizeof(after_line), "if %s goto %s", a.arg1, a.result);
            else if (strcmp(a.op, "label") == 0)
                snprintf(after_line, sizeof(after_line), "%s:", a.result);
            else
                snprintf(after_line, sizeof(after_line), "%s = %s %s %s", a.result, a.arg1, a.op, a.arg2);

            printf("%-5d %-38s %-38s\n", i, before_line, after_line);
            changed = 1;
        }
    }

    if (!changed) {
        printf("No TAC lines changed. No foldable constant expression was found.\n");
    }

    printf("================================================================\n");
}


static int is_temp_name(const char* s) {
    if (!s || s[0] != 't') return 0;
    if (s[1] < '0' || s[1] > '9') return 0;
    for (int i = 1; s[i]; i++) {
        if (s[i] < '0' || s[i] > '9') return 0;
    }
    return 1;
}

static int temp_used_after(int start, const char* temp) {
    for (int i = start; i < tac_count; i++) {
        if (strcmp(tac_table[i].arg1, temp) == 0 ||
            strcmp(tac_table[i].arg2, temp) == 0) {
            return 1;
        }
    }
    return 0;
}

static void remove_tac_at(int index) {
    if (index < 0 || index >= tac_count) return;
    for (int i = index; i < tac_count - 1; i++) {
        tac_table[i] = tac_table[i + 1];
        tac_table[i].index = i;
    }
    tac_count--;
}

void optimize_dead_temp_elimination(void) {
    for (int i = 0; i < tac_count; i++) {
        Quadruple* q = &tac_table[i];

        if (is_temp_name(q->result) &&
            (strcmp(q->op, "=") == 0 || is_arithmetic_op(q->op) ||
             is_relational_op(q->op) || is_logic_op(q->op) || strcmp(q->op, "!") == 0) &&
            !temp_used_after(i + 1, q->result)) {
            remove_tac_at(i);
            i--;
        }
    }
}

static void format_quad_readable(Quadruple q, char* out, int out_size) {
    if (strcmp(q.op, "=") == 0)
        snprintf(out, out_size, "%s = %s", q.result, q.arg1);
    else if (strcmp(q.op, "goto") == 0)
        snprintf(out, out_size, "goto %s", q.result);
    else if (strcmp(q.op, "ifTrue") == 0)
        snprintf(out, out_size, "if %s goto %s", q.arg1, q.result);
    else if (strcmp(q.op, "label") == 0)
        snprintf(out, out_size, "%s:", q.result);
    else if (strcmp(q.op, "print") == 0)
        snprintf(out, out_size, "print %s", q.arg1);
    else if (strcmp(q.op, "!") == 0)
        snprintf(out, out_size, "%s = !%s", q.result, q.arg1);
    else
        snprintf(out, out_size, "%s = %s %s %s", q.result, q.arg1, q.op, q.arg2);
}

void print_removed_tac_lines(Quadruple before[], int before_count) {
    int removed = 0;

    printf("\n================ DEAD CODE ELIMINATION REMOVED LINES ================\n");
    printf("%-5s %-50s %-20s\n", "Old", "Removed TAC", "Reason");
    printf("--------------------------------------------------------------------------------\n");

    for (int i = 0; i < before_count; i++) {
        int found = 0;
        for (int j = 0; j < tac_count; j++) {
            if (strcmp(before[i].op, tac_table[j].op) == 0 &&
                strcmp(before[i].arg1, tac_table[j].arg1) == 0 &&
                strcmp(before[i].arg2, tac_table[j].arg2) == 0 &&
                strcmp(before[i].result, tac_table[j].result) == 0) {
                found = 1;
                break;
            }
        }

        if (!found) {
            char line[512];
            format_quad_readable(before[i], line, sizeof(line));
            printf("%-5d %-50s %-20s\n", i, line, "unused temporary");
            removed = 1;
        }
    }

    if (!removed) {
        printf("No dead temporary assignments were removed.\n");
    }
    printf("=====================================================================\n");
}


void print_optimization_explanation(void) {
    printf("\n================ OPTIMIZATION EXPLANATION ================\n");
    printf("Technique 1: Constant Folding.\n");
    printf("Meaning: if both operands of an operation are constants, the compiler evaluates it at compile time.\n");
    printf("Example: t1 = 2 * 3 becomes t1 = 6, and later uses of t1 are replaced by 6.\n\n");
    printf("Technique 2: Dead Code Elimination for unused temporaries.\n");
    printf("Meaning: if a temporary value is computed but never used later, that TAC line is removed.\n");
    printf("Example: after x = 14, unused temporary assignments like t1 = 12 can be removed.\n");
    printf("Both optimizations preserve semantics because they do not change final variable values or control flow.\n");
    printf("==========================================================\n");
}
static void asm_load_operand(const char* operand) {
    printf("    LOAD R1, %s\n", operand);
}

void print_target_code(void) {
    printf("\n================ TARGET CODE / PSEUDO ASSEMBLY ================\n");
    printf("; Generated directly from optimized TAC quadruples\n");

    for (int i = 0; i < tac_count; i++) {
        Quadruple* q = &tac_table[i];

        if (strcmp(q->op, "label") == 0) {
            printf("%s:\n", q->result);
        }
        else if (strcmp(q->op, "=") == 0) {
            printf("    MOV %s, %s\n", q->result, q->arg1);
        }
        else if (strcmp(q->op, "+") == 0) {
            asm_load_operand(q->arg1);
            printf("    ADD R1, %s\n", q->arg2);
            printf("    STORE %s, R1\n", q->result);
        }
        else if (strcmp(q->op, "-") == 0) {
            asm_load_operand(q->arg1);
            printf("    SUB R1, %s\n", q->arg2);
            printf("    STORE %s, R1\n", q->result);
        }
        else if (strcmp(q->op, "*") == 0) {
            asm_load_operand(q->arg1);
            printf("    MUL R1, %s\n", q->arg2);
            printf("    STORE %s, R1\n", q->result);
        }
        else if (strcmp(q->op, "/") == 0) {
            asm_load_operand(q->arg1);
            printf("    DIV R1, %s\n", q->arg2);
            printf("    STORE %s, R1\n", q->result);
        }
        else if (strcmp(q->op, "%") == 0) {
            asm_load_operand(q->arg1);
            printf("    MOD R1, %s\n", q->arg2);
            printf("    STORE %s, R1\n", q->result);
        }
        else if (is_relational_op(q->op)) {
            printf("    CMP %s, %s\n", q->arg1, q->arg2);
            printf("    SET%s %s\n", q->op, q->result);
        }
        else if (strcmp(q->op, "&&") == 0) {
            asm_load_operand(q->arg1);
            printf("    AND R1, %s\n", q->arg2);
            printf("    STORE %s, R1\n", q->result);
        }
        else if (strcmp(q->op, "||") == 0) {
            asm_load_operand(q->arg1);
            printf("    OR R1, %s\n", q->arg2);
            printf("    STORE %s, R1\n", q->result);
        }
        else if (strcmp(q->op, "!") == 0) {
            asm_load_operand(q->arg1);
            printf("    NOT R1\n");
            printf("    STORE %s, R1\n", q->result);
        }
        else if (strcmp(q->op, "ifTrue") == 0) {
            printf("    CMP %s, 0\n", q->arg1);
            printf("    JNE %s\n", q->result);
        }
        else if (strcmp(q->op, "goto") == 0) {
            printf("    JMP %s\n", q->result);
        }
        else if (strcmp(q->op, "print") == 0) {
            printf("    PRINT %s\n", q->arg1);
        }
        else {
            printf("    ; Unsupported TAC: (%s, %s, %s, %s)\n", q->op, q->arg1, q->arg2, q->result);
        }
    }

    printf("===============================================================\n");
}

void print_target_code_explanation(void) {
    printf("\n================ TARGET CODE EXPLANATION ================\n");
    printf("Arithmetic TAC is translated using LOAD, arithmetic instruction, and STORE.\n");
    printf("Example: t1 = a + b becomes LOAD R1, a; ADD R1, b; STORE t1, R1.\n");
    printf("Control flow uses labels and jumps. ifTrue cond goto L becomes CMP cond, 0 followed by JNE L.\n");
    printf("while loops are represented by a start label, condition check, body label, back jump, and exit label.\n");
    printf("The target code is generated from the optimized TAC, so semantics are preserved.\n");
    printf("=========================================================\n");
}

/* =================== LL(1) PARSER PART =================== */

/* LL(1) grammar (expressions only)
   E  -> T E'
   E' -> OP_ADD T E' | epsilon
   T  -> F T'
   T' -> OP_MUL F T' | epsilon
   F  -> ID | INT_CONST | FLOAT_CONST | '(' E ')'
*/

typedef enum {
    LL_ID = 1,
    LL_INT_CONST,
    LL_FLOAT_CONST,
    LL_OP_ADD,
    LL_OP_MUL,
    LL_LPAREN,
    LL_RPAREN,
    LL_SEMI,
    LL_END
} LLTokenType;

typedef struct {
    LLTokenType type;
    char        lexeme[64];
} LLToken;

static LLToken ll_look;

static LLToken ll_next_token(void);
static void ll_advance(void);
static void ll_error(const char* where, const char* expected);
static void parse_E(void);
static void parse_Ep(void);
static void parse_T(void);
static void parse_Tp(void);
static void parse_F(void);

static void print_first_follow(void) {
    printf("\n==== FIRST / FOLLOW for LL(1) Expression Grammar ====\n");
    printf("FIRST(E)  = { ID, INT_CONST, FLOAT_CONST, '(' }\n");
    printf("FIRST(E') = { OP_ADD, epsilon }\n");
    printf("FIRST(T)  = { ID, INT_CONST, FLOAT_CONST, '(' }\n");
    printf("FIRST(T') = { OP_MUL, epsilon }\n");
    printf("FIRST(F)  = { ID, INT_CONST, FLOAT_CONST, '(' }\n\n");

    printf("FOLLOW(E)  = { ')', ';', EOF }\n");
    printf("FOLLOW(E') = { ')', ';', EOF }\n");
    printf("FOLLOW(T)  = { OP_ADD, ')', ';', EOF }\n");
    printf("FOLLOW(T') = { OP_ADD, ')', ';', EOF }\n");
    printf("FOLLOW(F)  = { OP_MUL, OP_ADD, ')', ';', EOF }\n\n");
}

static LLToken ll_next_token(void) {
    LLToken t;
    int tk = yylex();
    t.lexeme[0] = '\0';

    switch (tk) {
    case ID:
        t.type = LL_ID; strncpy(t.lexeme, yytext, 63); break;
    case INT_CONST:
        t.type = LL_INT_CONST; strncpy(t.lexeme, yytext, 63); break;
    case FLOAT_CONST:
        t.type = LL_FLOAT_CONST; strncpy(t.lexeme, yytext, 63); break;
    case OP_ADD:
        t.type = LL_OP_ADD; strncpy(t.lexeme, yytext, 63); break;
    case OP_MUL:
        t.type = LL_OP_MUL; strncpy(t.lexeme, yytext, 63); break;
    case '(':
        t.type = LL_LPAREN; strcpy(t.lexeme, "("); break;
    case ')':
        t.type = LL_RPAREN; strcpy(t.lexeme, ")"); break;
    case ';':
        t.type = LL_SEMI; strcpy(t.lexeme, ";"); break;
    case 0:
        t.type = LL_END; strcpy(t.lexeme, "EOF"); break;
    default:
        t.type = LL_END; strcpy(t.lexeme, "EOF"); break;
    }

    t.lexeme[63] = '\0';
    return t;
}

static void ll_advance(void) {
    ll_look = ll_next_token();
}

static const char* ll_tok_name(LLTokenType t) {
    switch (t) {
    case LL_ID:          return "id";
    case LL_INT_CONST:   return "INT_CONST";
    case LL_FLOAT_CONST: return "FLOAT_CONST";
    case LL_OP_ADD:      return "OP_ADD(+/-)";
    case LL_OP_MUL:      return "OP_MUL(*//%)";
    case LL_LPAREN:      return "'('";
    case LL_RPAREN:      return "')'";
    case LL_SEMI:        return "';'";
    case LL_END:         return "EOF";
    default:             return "?";
    }
}

static void print_ll_header(void) {
    printf("====================================================================================================================\n");
    printf("| %-20s | %-40s | %-40s |\n",
           "Stack / Nonterminal", "Current Input (lookahead)", "Parsing Action");
    printf("====================================================================================================================\n");
}

static void print_ll_row(const char* stack, const char* action) {
    char inputbuf[80];
    snprintf(inputbuf, sizeof(inputbuf), "%s (\"%s\")",
             ll_tok_name(ll_look.type), ll_look.lexeme);
    printf("| %-20s | %-40s | %-40s |\n", stack, inputbuf, action);
}

static void ll_error(const char* where, const char* expected) {
    fprintf(stderr,
            "[LL(1) error] %s : expected %s, got %s ('%s') at line %d\n",
            where, expected,
            ll_tok_name(ll_look.type),
            ll_look.lexeme,
            line_number);
    exit(1);
}

static void parse_E(void) {
    print_ll_row("E", "E -> T E'");
    parse_T();
    parse_Ep();
}

static void parse_Ep(void) {
    if (ll_look.type == LL_OP_ADD) {
        print_ll_row("E'", "E' -> OP_ADD T E'");
        ll_advance();
        parse_T();
        parse_Ep();
    } else if (ll_look.type == LL_RPAREN ||
               ll_look.type == LL_SEMI   ||
               ll_look.type == LL_END) {
        print_ll_row("E'", "E' -> epsilon");
    } else {
        print_ll_row("E'", "ERROR");
        ll_error("E'", "OP_ADD or FOLLOW(E')");
    }
}

static void parse_T(void) {
    print_ll_row("T", "T -> F T'");
    parse_F();
    parse_Tp();
}

static void parse_Tp(void) {
    if (ll_look.type == LL_OP_MUL) {
        print_ll_row("T'", "T' -> OP_MUL F T'");
        ll_advance();
        parse_F();
        parse_Tp();
    } else if (ll_look.type == LL_OP_ADD   ||
               ll_look.type == LL_RPAREN   ||
               ll_look.type == LL_SEMI     ||
               ll_look.type == LL_END) {
        print_ll_row("T'", "T' -> epsilon");
    } else {
        print_ll_row("T'", "ERROR");
        ll_error("T'", "OP_MUL or FOLLOW(T')");
    }
}

static void parse_F(void) {
    if (ll_look.type == LL_ID) {
        print_ll_row("F", "F -> ID");
        ll_advance();
    } else if (ll_look.type == LL_INT_CONST) {
        print_ll_row("F", "F -> INT_CONST");
        ll_advance();
    } else if (ll_look.type == LL_FLOAT_CONST) {
        print_ll_row("F", "F -> FLOAT_CONST");
        ll_advance();
    } else if (ll_look.type == LL_LPAREN) {
        print_ll_row("F", "F -> ( E )");
        ll_advance();
        parse_E();
        if (ll_look.type != LL_RPAREN)
            ll_error("F", "')'");
        ll_advance();
    } else {
        print_ll_row("F", "ERROR");
        ll_error("F", "ID/INT_CONST/FLOAT_CONST/'('");
    }
}

static void run_ll1_demo(void) {
    print_first_follow();
    printf("Enter an arithmetic expression terminated by ';'\n");
    printf("Example: a + 3 * (b - 2);\n\n");

    ll_advance();

    print_ll_header();
    parse_E();

    if (ll_look.type != LL_SEMI)
        ll_error("run_ll1_demo", "';'");
    print_ll_row("match", "match ';' and ACCEPT");
    printf("====================================================================================================================\n");
    printf("\nLL(1) parser: expression accepted.\n");
}

/* ================= SHIFT-REDUCE TABLE DEMO ================= */

typedef struct {
    char items[200][32];
    int top;
} SRStack;

static void sr_init(SRStack* s) {
    s->top = -1;
}

static void sr_push(SRStack* s, const char* x) {
    if (s->top < 199) {
        s->top++;
        strncpy(s->items[s->top], x, 31);
        s->items[s->top][31] = '\0';
    }
}

static void sr_pop(SRStack* s, int k) {
    s->top -= k;
    if (s->top < -1) s->top = -1;
}

static void sr_stack_string(SRStack* s, char* out) {
    out[0] = '\0';
    strcat(out, "$");
    for (int i = 0; i <= s->top; i++) {
        strcat(out, " ");
        strcat(out, s->items[i]);
    }
}

static void sr_input_string(char tokens[][32], int n, int pos, char* out) {
    out[0] = '\0';
    for (int i = pos; i < n; i++) {
        strcat(out, tokens[i]);
        if (i != n - 1) strcat(out, " ");
    }
    if (strlen(out) > 0) strcat(out, " ");
    strcat(out, "$");
}

static void sr_print_header(void) {
    printf("\n================================================================================\n");
    printf("| %-28s | %-26s | %-27s |\n", "Stack", "Input", "Action");
    printf("================================================================================\n");
}

static void sr_print_row(SRStack* st, char tokens[][32], int n, int pos, const char* action) {
    char stackbuf[512], inputbuf[512];
    sr_stack_string(st, stackbuf);
    sr_input_string(tokens, n, pos, inputbuf);
    printf("| %-28s | %-26s | %-27s |\n", stackbuf, inputbuf, action);
}

static const char* sr_next_input(char tokens[][32], int n, int pos) {
    if (pos < n) return tokens[pos];
    return "$";
}

static int sr_try_reduce(SRStack* st, char tokens[][32], int n, int pos) {
    const char* next = sr_next_input(tokens, n, pos);

    if (st->top >= 0 &&
        (!strcmp(st->items[st->top], "id") || !strcmp(st->items[st->top], "num"))) {

        char rhs[32];
        strcpy(rhs, st->items[st->top]);

        sr_pop(st, 1);
        sr_push(st, "F");

        char action[64];
        snprintf(action, sizeof(action), "Reduce F -> %s", rhs);
        sr_print_row(st, tokens, n, pos, action);
        return 1;
    }

    if (st->top >= 2 &&
        !strcmp(st->items[st->top - 2], "(") &&
        !strcmp(st->items[st->top - 1], "E") &&
        !strcmp(st->items[st->top], ")")) {

        sr_pop(st, 3);
        sr_push(st, "F");

        sr_print_row(st, tokens, n, pos, "Reduce F -> ( E )");
        return 1;
    }

    if (st->top >= 2 &&
        !strcmp(st->items[st->top - 2], "T") &&
        !strcmp(st->items[st->top - 1], "*") &&
        !strcmp(st->items[st->top], "F")) {

        sr_pop(st, 3);
        sr_push(st, "T");

        sr_print_row(st, tokens, n, pos, "Reduce T -> T * F");
        return 1;
    }

    if (st->top >= 0 && !strcmp(st->items[st->top], "F")) {

        sr_pop(st, 1);
        sr_push(st, "T");

        sr_print_row(st, tokens, n, pos, "Reduce T -> F");
        return 1;
    }

    if (st->top >= 2 &&
        !strcmp(st->items[st->top - 2], "E") &&
        !strcmp(st->items[st->top - 1], "+") &&
        !strcmp(st->items[st->top], "T")) {

        if (strcmp(next, "*") != 0) {
            sr_pop(st, 3);
            sr_push(st, "E");

            sr_print_row(st, tokens, n, pos, "Reduce E -> E + T");
            return 1;
        }
    }

    if (st->top >= 0 && !strcmp(st->items[st->top], "T")) {

        if (strcmp(next, "*") != 0) {
            sr_pop(st, 1);
            sr_push(st, "E");

            sr_print_row(st, tokens, n, pos, "Reduce E -> T");
            return 1;
        }
    }

    return 0;
}

static int sr_tokenize_expr(const char* line, char tokens[][32]) {
    int i = 0, j = 0, n = 0;

    while (line[i]) {
        if (line[i] == ' ' || line[i] == '\t' || line[i] == '\n') {
            i++;
            continue;
        }

        if ((line[i] >= 'a' && line[i] <= 'z') ||
            (line[i] >= 'A' && line[i] <= 'Z') ||
            line[i] == '_') {
            j = 0;
            while ((line[i] >= 'a' && line[i] <= 'z') ||
                   (line[i] >= 'A' && line[i] <= 'Z') ||
                   (line[i] >= '0' && line[i] <= '9') ||
                   line[i] == '_') {
                if (j < 30) tokens[n][j++] = line[i];
                i++;
            }
            tokens[n][j] = '\0';
            strcpy(tokens[n], "id");
            n++;
            continue;
        }

        if (line[i] >= '0' && line[i] <= '9') {
            j = 0;
            while ((line[i] >= '0' && line[i] <= '9') || line[i] == '.') {
                if (j < 30) tokens[n][j++] = line[i];
                i++;
            }
            tokens[n][j] = '\0';
            strcpy(tokens[n], "num");
            n++;
            continue;
        }

        if (line[i] == '+' || line[i] == '*' || line[i] == '(' || line[i] == ')') {
            tokens[n][0] = line[i];
            tokens[n][1] = '\0';
            n++;
            i++;
            continue;
        }

        if (line[i] == ';') break;

        i++;
    }

    return n;
}

static void run_shift_reduce_table_demo(void) {
    char line[512];
    char tokens[200][32];
    int n, pos = 0;
    SRStack st;

    printf("\nEnter an expression for shift-reduce table demo:\n");
    printf("Examples:\n");
    printf("  a + 3 * b;\n");
    printf("  2 * (3 + 4);\n\n");

    if (!fgets(line, sizeof(line), stdin)) {
        printf("No input.\n");
        return;
    }

    n = sr_tokenize_expr(line, tokens);
    sr_init(&st);

    sr_print_header();

    while (pos < n) {
        sr_push(&st, tokens[pos]);

        char action[64];
        snprintf(action, sizeof(action), "Shift %s", tokens[pos]);
        sr_print_row(&st, tokens, n, pos + 1, action);
        pos++;

        while (sr_try_reduce(&st, tokens, n, pos)) {
            ;
        }
    }

    while (sr_try_reduce(&st, tokens, n, pos)) {
        ;
    }

    if (st.top == 0 && !strcmp(st.items[0], "E")) {
        sr_print_row(&st, tokens, n, pos, "Accept");
    } else {
        sr_print_row(&st, tokens, n, pos, "Reject");
    }

    printf("================================================================================\n");
}
